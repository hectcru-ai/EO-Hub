function handlePhotoUpload(kit, file, caption){

  if(!file) return;

  const reader = new FileReader();

  reader.onload = (e)=>{

    const img = new Image();

    img.onload = async ()=>{

      // Try progressively smaller/more compressed versions rather than

      // failing on the first attempt — a real, detailed photo can easily

      // exceed the per-photo limit at the initial size, and most photos

      // still look perfectly fine as a kit reference at a smaller size.

      const attempts = [

        {maxW: 650, quality: 0.65},

        {maxW: 500, quality: 0.55},

        {maxW: 400, quality: 0.45},

        {maxW: 320, quality: 0.4},

        {maxW: 260, quality: 0.35},

      ];

      let dataUrl = null;

      for(const {maxW, quality} of attempts){

        const scale = Math.min(1, maxW / img.width);

        const canvas = document.createElement('canvas');

        canvas.width = img.width * scale;

        canvas.height = img.height * scale;

        const ctx = canvas.getContext('2d');

        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);

        const candidate = canvas.toDataURL('image/jpeg', quality);

        if(candidate.length <= 190000){ dataUrl = candidate; break; }

      }

      if(!dataUrl){

        alert('⚠ This photo is unusually large even after compression — try a different photo, or a screenshot of it instead.');

        return;

      }

      try{

        await addKitPhoto(kit.id, dataUrl, caption);

        await setKitMediaStatus(kit.id, 'photo', true);

        if(!kitMediaIndex[kit.id]) kitMediaIndex[kit.id] = {photo:false, video:false};

        kitMediaIndex[kit.id].photo = true;

      }catch(err){

        alert('⚠ ' + err.message);

        return;

      }

      loadPhotoPane(kit);

      refreshMediaBadgeOnly(kit.id);

    };

    img.src = e.target.result;

  };

  reader.readAsDataURL(file);

}

// YouTube's own "Embed" button gives you a full <iframe> HTML snippet to

// copy, not a plain link — pasting that whole snippet in used to break the

// page, since the quotes inside it closed our own src="..." early and the

// rest rendered as literal text. This pulls the real URL out either way.

// Wraps matches of `term` in <mark> — safe against HTML in `html`, since it

// only touches text between tags, never the tags/attributes themselves, so a

// search term can never corrupt markup even if it happens to match a class

// name or similar. Works fine on plain text too (no tags = one text segment).

function highlightMatch(html, term){

  if(!term) return html;

  const escaped = term.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');

  const re = new RegExp(`(${escaped})`, 'gi');

  return html.split(/(<[^>]+>)/g).map(part =>

    part.startsWith('<') ? part : part.replace(re, '<mark>$1</mark>')

  ).join('');

}


function extractVideoUrl(input){

  const iframeMatch = input.match(/<iframe[^>]*\ssrc=["']([^"']+)["']/i);

  return iframeMatch ? iframeMatch[1] : input;

}


// Defense-in-depth: even a clean-looking URL could contain a stray quote

// character, which would otherwise break out of the src="..."/href="..."

// attribute it's inserted into and render as literal text on the page.

function escapeHtmlAttr(str){

  return String(str).replace(/&/g, '&amp;').replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;');

}

  return newOrderKits.map(inst => ALL_KITS.find(k=>String(k.id)===String(inst.kitId))).filter(k=>k)

    .map(k => eligibility(k, newOrderStudentGrade)).filter(e => e.status === 'blocked');

}


async function saveNewOrder(){

  const statusEl = document.getElementById('saveStatusMsg');

  if(!requireStaffAuth('save a checkout record')) return;

  // Check condition log gate

  const openIssuesForOrder = newOrderKits.some(inst =>

    openConditionLogs.some(e => String(e.kitId) === String(inst.kitId))

  );

  if(openIssuesForOrder && !conditionLogAck){

    statusEl.textContent = '⚠ This order has open condition log issues that must be reviewed and acknowledged before saving. Check the box at the top of the checklist.';

    statusEl.style.color = 'var(--danger)'; return;

  }

  const mismatches = getGradeMismatches();

  if(mismatches.length && !newOrderOverride.ack){

    statusEl.textContent = '⚠ This order has a grade-eligibility conflict that hasn\'t been acknowledged. Check the override box above, or remove the flagged kit(s), before saving.';

    statusEl.style.color = 'var(--danger)'; return;

  }

  if(mismatches.length && newOrderOverride.ack && !newOrderOverride.reason.trim()){

    statusEl.textContent = '⚠ An override reason is required when overriding a grade-eligibility conflict.';

    statusEl.style.color = 'var(--danger)'; return;

  }

  const order = buildOrderObjectDraft();

  if(order.reservationNumber === '(no reservation #)'){

    statusEl.textContent = '⚠ Please enter a reservation number before saving.'; statusEl.style.color = 'var(--danger)'; return;

  }

  if(!order.checkoutSignatureStudent){

    statusEl.textContent = '⚠ Student signature (typed name) is required before saving.'; statusEl.style.color = 'var(--danger)'; return;

  }

  if(!order.checkoutSignatureStaff){

    statusEl.textContent = '⚠ Staff signature (typed name) is required before saving.'; statusEl.style.color = 'var(--danger)'; return;

  }

  try{

    await saveOrder(order);

    statusEl.textContent = `✅ Saved. Staff can look this up at check-in using reservation #${order.reservationNumber}.`;

    statusEl.style.color = 'var(--success)';

    // Reset all state so the form is clean for the next checkout

    newOrderKits = [];

    newOrderInstanceCounter = 0;

    newOrderStudentGrade = '';

    newOrderOverride = {ack:false, reason:''};

    newOrderResNum = '';

    newOrderStudentName = '';

    newOrderStudentSig = '';

    newOrderStaffSig = '';

    newOrderChecklist = {};

    conditionLogAck = false;

    setTimeout(()=>{ renderNewOrderPane(); }, 1800); // brief pause so staff can read the confirmation

  }catch(err){

    statusEl.textContent = '⚠ Could not save — ' + err.message + '. Try printing as a backup.';

    statusEl.style.color = 'var(--danger)';

  }

}

    statusEl.style.color = 'var(--danger)'; return;

  }

  if(mismatches.length && newOrderOverride.ack && !newOrderOverride.reason.trim()){

    statusEl.textContent = '⚠ An override reason is required when overriding a grade-eligibility conflict.';

// The actual check-in workflow — staff re-verify each item against what's

// physically coming back, not just a static replay of what checkout said.

// Unlike the checkout-time checklist, nothing pre-checks itself here:

// every item starts exactly as it was left at checkout, and staff mark it

// off (or flag it) based on what's actually in front of them right now.

// Matches the actual fee categories and language from the Policy Manual —

// not invented figures. Fixed-fee items compute automatically; anything

// needing a real replacement-cost lookup (no price database here) gets a

// manual $ field instead of a guess.

const CHECKIN_ISSUE_TYPES = {

  '': { label: 'No issue', needsCost: false },

  'missing': { label: 'Missing', needsCost: false,

    email: (item) => `${item.item} was not returned. Per policy, an item is tracked as missing for 7 calendar days from the due date — if it isn't returned or located within that window, it's considered lost and the full replacement cost will be charged instead of ongoing late fees.` },

  'damaged_repairable': { label: 'Damaged (operable / repairable in-house)', needsCost: false, fixedFee: 30,

    email: (item) => `${item.item} was returned damaged. Per policy, the standard damage fee for an item that's still operable or repairable in-house is $30.00.` },

  'damaged_inoperable': { label: 'Damaged (not repairable in-house)', needsCost: true,

    email: (item, cost) => `${item.item} was returned damaged beyond in-house repair. Per policy, this means the full replacement cost applies${cost ? `: $${cost}` : ' (amount to be confirmed by the Manager)'}.` },

  'dirty': { label: 'Dirty', needsCost: false, fixedFee: 10,

    email: (item) => `${item.item} was returned dirty (smoke, dirt, chalk, sand, or fingerprints). Per policy, the cleaning fee is $10.00.` },

  'scratched_lens': { label: 'Scratched Lens', needsCost: true,

    email: (item, cost) => `${item.item} (lens) was returned scratched. Per policy, a scratched lens is considered unrepairable, and the replacement cost at used price applies${cost ? `: $${cost}` : ' (amount to be confirmed by the Manager)'}.` },

  'wet': { label: 'Wet / Waterlogged', needsCost: true,

    email: (item, cost) => `${item.item} was returned wet/waterlogged. Per policy, this is evaluated case-by-case and can go up to the full replacement cost${cost ? `: $${cost}` : ' (amount to be confirmed by the Manager)'}.` },

};


function computeLateFee(order){

  if(!order.lateReturn || !order.daysLate) return 0;

  const totalItems = (order.kits||[]).reduce((sum,k) => sum + (k.items||[]).length, 0);

  const dailyFee = Math.min(5 * totalItems, 50);

  return dailyFee * order.daysLate;

}


function generateInvoiceNumber(resNum){

  const d = new Date();

  const stamp = `${d.getFullYear()}${String(d.getMonth()+1).padStart(2,'0')}${String(d.getDate()).padStart(2,'0')}`;

  return `INV-${resNum}-${stamp}`;

}


function invoiceLine(description, amount){

  const desc = amount !== null ? description + ' ' : description;

  const amountStr = amount !== null ? `$${amount.toFixed(2)}` : '';

  if(amount === null) return desc;

  const dots = Math.max(3, 60 - desc.length - amountStr.length);

  return desc + '.'.repeat(dots) + ' ' + amountStr;

}


function buildCheckInEmailDraft(order){

  const chargeLines = []; // [{desc, note, amount|null}]

  let hasFixedTotal = true;

  let total = 0;


  const lateFee = computeLateFee(order);

  if(lateFee > 0){

    chargeLines.push({

      desc: `Late return (${order.daysLate} day${order.daysLate==1?'':'s'})`,

      note: 'Per policy: $5.00/item/day, capped at $50.00/day combined.',

      amount: lateFee,

    });

    total += lateFee;

  }


  (order.kits||[]).forEach(k=>{

    (k.items||[]).forEach(it=>{

      if(!it.issueType) return;


      if(it.issueType === 'missing' && it.confirmedLost){

        const amount = parseFloat(it.replacementCost) || 0;

        total += amount;

        chargeLines.push({

          desc: `${it.item} — Confirmed Lost`,

          note: 'Per policy: the 7-day missing window closed without the item being found or returned — full replacement cost now applies.',

          amount,

        });

        return;

      }


      const def = CHECKIN_ISSUE_TYPES[it.issueType];

      if(!def || !def.email) return;

      let amount = null;

      if(def.fixedFee){ amount = def.fixedFee; total += amount; }

      else if(def.needsCost){

        if(it.replacementCost){ amount = parseFloat(it.replacementCost) || 0; total += amount; }

        else hasFixedTotal = false;

      }

      chargeLines.push({

        desc: `${it.item} — ${def.label}`,

        note: def.email(it, it.replacementCost).replace(/^.*?Per policy/, 'Per policy'),

        amount,

      });

    });

  });


  if(!chargeLines.length) return null;


  const invoiceNumber = generateInvoiceNumber(order.reservationNumber);

  const dateStr = new Date().toLocaleDateString('en-US', {year:'numeric', month:'long', day:'numeric'});

  const totalLabel = hasFixedTotal ? 'TOTAL' : 'TOTAL (partial — some amounts still TBD)';


  const subject = `[DRAFT — for review] Invoice ${invoiceNumber} — ${order.studentName}`;

  const body =

`INVOICE (DRAFT — FOR INTERNAL REVIEW, NOT YET SENT TO STUDENT)


Invoice #: ${invoiceNumber}

Date: ${dateStr}

Order: ${order.reservationNumber}

Student: ${order.studentName}


CHARGES

${'-'.repeat(60)}

${chargeLines.map(l => invoiceLine(l.desc, l.amount) + `\n  ${l.note}`).join('\n\n')}

${'-'.repeat(60)}

${invoiceLine(totalLabel, total)}


Payment: Business Office (busoff@regent.edu). Unpaid fees suspend Cheqroom access immediately; an academic hold is placed if unpaid by the end of the semester.


Generated by the EO Hub — review with Ivy before anything goes to the student.`;


  return { subject, body };

}

function downloadEmbeddedFile(base64Data, filename, mimeType){

  try{

    const byteChars = atob(base64Data);

    const byteNumbers = new Array(byteChars.length);

    for(let i = 0; i < byteChars.length; i++){ byteNumbers[i] = byteChars.charCodeAt(i); }

    const byteArray = new Uint8Array(byteNumbers);

    const blob = new Blob([byteArray], {type: mimeType});

    const url = URL.createObjectURL(blob);

    const a = document.createElement('a');

    a.href = url; a.download = filename;

    document.body.appendChild(a); a.click(); document.body.removeChild(a);

    setTimeout(()=>URL.revokeObjectURL(url), 1000);

  }catch(e){

    alert('⚠ Could not download this file. Try again, or ask for a fresh copy.');

  }

}

            <label for="ms-${m.key}">${m.label}</label>

          </div>`;

        }).join('')}

      </div>

    `).join('');

    content.querySelectorAll('[data-milestone-key]').forEach(cb=>{

      cb.addEventListener('change', async (e)=>{

        if(!requireStaffAuth('update a trainee\'s progress')){ e.target.checked = !e.target.checked; return; }

        const key = e.target.dataset.milestoneKey;

        const newVal = e.target.checked;

        try{

          await setTraineeMilestone(currentTraineeId, key, newVal);

          currentTraineeData.milestones[key] = newVal;

          renderTraineeTracker();

        }catch(err){

          e.target.checked = !newVal;

          alert('⚠ Could not save — ' + err.message);

        }

      });

    });

  } else {

    const sortedKits = [...ALL_KITS].sort((a,b)=>a.name.localeCompare(b.name));

    content.innerHTML = `<div class="kit-mastery-grid">${sortedKits.map(k => {

      const done = !!kits[k.id];

      return `<div class="kit-mastery-item ${done?'mastered':''}">

        <input type="checkbox" id="km-${k.id}" data-kit-key="${k.id}" ${done?'checked':''}>

        <label for="km-${k.id}">${k.name}</label>

      </div>`;

    }).join('')}</div>`;

    content.querySelectorAll('[data-kit-key]').forEach(cb=>{

      cb.addEventListener('change', async (e)=>{

        if(!requireStaffAuth('update a trainee\'s progress')){ e.target.checked = !e.target.checked; return; }

        const key = e.target.dataset.kitKey;

        const newVal = e.target.checked;

        try{

          await setTraineeKitMastery(currentTraineeId, key, newVal);

          currentTraineeData.kits[key] = newVal;

          renderTraineeTracker();

        }catch(err){

          e.target.checked = !newVal;

          alert('⚠ Could not save — ' + err.message);

        }

      });

    });

  }

}
